import torch
from PIL import Image
from transformers import AutoModel, AutoTokenizer

if __name__ == '__main__':
    prompt = f"""### 背景 ###
        您需要对图片中的内容进行识别。
        并大致确定它们在图中的位置。烟雾弹、弹夹、军用手电筒、手雷、剩余的其他需要识别出的物品（务必推测其物体具体是什么）
        要求输出格式
        {{
          "id": "1",
          "image": "/root/data/1.png",
          "conversations": [
            {{
              "role": "user",
              "content": "图中机械臂托盘里有哪些物品？"
            }},
            {{
              "role": "assistant",
              "content": "烟雾弹、弹夹、军用手电筒、手雷，以及另一枚横放的烟雾弹。"
            }},
            {{
              "role": "user",
              "content": "手雷在托盘的什么位置？"
            }},
            {{
              "role": "assistant",
              "content": "右上角附近。"
            }}
            ...
          ]
        }}
    """

    model_file = '/root/gpufree-data/FM9G4B-V'
    model = AutoModel.from_pretrained(model_file, trust_remote_code=True,
        attn_implementation='sdpa', torch_dtype=torch.bfloat16)
    model = model.eval().cuda()
    tokenizer = AutoTokenizer.from_pretrained(model_file, trust_remote_code=True)

    image = Image.open('data/1.png').convert('RGB')

    msgs = [{'role': 'user', 'content': [image, prompt]}]

    res = model.chat(
        image=None,
        msgs=msgs,
        tokenizer=tokenizer
    )
    print("\n", "="*100, "\n")
    print(res)


    # 第二轮聊天，传递多轮对话的历史信息
    msgs.append({"role": "assistant", "content": [res]})
    msgs.append({"role": "user", "content": ["对图片中的内容进行识别。并大致确定它们在图中的位置。烟雾弹、弹夹、军用手电筒、手雷、剩余的其他需要识别出的物品（务必推测其物体具体是什么），并按照输出格式输出"]})

    answer = model.chat(
        image=None,
        msgs=msgs,
        tokenizer=tokenizer
    )
    print("\n", "="*100, "\n")
    print(answer)


    ## 流式输出，设置：
    # sampling=True
    # stream=True
    ## 返回一个生成器
    msgs = [{'role': 'user', 'content': [image, prompt]}]
    res = model.chat(
        image=None,
        msgs=msgs,
        tokenizer=tokenizer,
        sampling=True,
        stream=True
    )
    print("\n", "="*100, "\n")
    generated_text = ""
    for new_text in res:
        generated_text += new_text
        print(new_text, flush=True, end='')

3.2. 微调训练

3.2.1 主流PEFT (Parameter-Efficient Fine-Tuning, 参数高效微调) 方法介绍

方法



核心思想

优点

缺点

LoRA





在权重矩阵旁并联一个低秩小矩阵，只训练小矩阵。



效果好、速度快、通用性强、社区支持最广。权重文件小（MB级）。

相比全量微调，对显存的节省有限（因为原模型仍需以16位加载）。

QLoRA







量化 (Quantization) + LoRA。先将大模型权重量化成4位来大幅减少显存占用，然后在这4位的模型上应用LoRA进行训练。

极大降低显存需求，可以在消费级显卡（如24GB/40GB）上微调非常大的模型（如70B）。训练效果几乎与16位LoRA持平。**是当前微调大模型的黄金标准**。

训练速度比标准LoRA稍慢一些（因为有量化/反量化的计算开销）。对极小模型效果可能不如LoRA。

Prefix-Tuning







在输入前加入可训练的“虚拟Token”（前缀）。

对生成式任务效果好。不改变原模型结构。

训练可能不稳定，超参数难调。性能上限通常不如LoRA。

P-Tuning (v2)





Prefix-Tuning的改进版，在模型的每一层都加入虚拟Token。

在NLU任务上表现优异，尤其是在中小模型上。



配置比LoRA复杂，通用性稍弱。

比赛方提供了LoRA微调相关代码

3.2.1 LoRA 训练

LoRA方法是通过在模型关键层（如注意力层）插入低秩矩阵，仅训练这些低秩矩阵，冻结预训练模型其他参数的方法实现预训练模型对下游任务的适配，官方提供原生LoRA微调相关脚本，可以满足在24GB显卡上的微调需求。

LoRA 参数

参数

作用

建议值范围

lora_r

LoRA 低秩维度，控制模型容量

4~64

lora_alpha

缩放因子，影响参数更新幅度

通常为lora_r的 2 倍

lora_dropout

防止过拟合的 dropout 率

0~0.1

lora_target_modules

插入LoRA层的模块



 q_lora

是否启用 QLoRA

True / False



Bash
finetune_lora.sh
# #!/bin/bash
GPUS_PER_NODE=1
NNODES=1
NODE_RANK=0
MASTER_ADDR=localhost
MASTER_PORT=6012

MODEL="xxxx/FM9G4B-V"
DATA="FM9G4B-V/finetune/data.json"
EVAL_DATA="FM9G4B-V/finetune/data.json"

MODEL_MAX_Length=2048 # if conduct multi-images sft, please set MODEL_MAX_Length=4096

DISTRIBUTED_ARGS="
    --nproc_per_node $GPUS_PER_NODE \
    --nnodes $NNODES \
    --node_rank $NODE_RANK \
    --master_addr $MASTER_ADDR \
    --master_port $MASTER_PORT
"
torchrun $DISTRIBUTED_ARGS finetune.py  \
    --model_name_or_path $MODEL \
    --data_path $DATA \
    --eval_data_path $EVAL_DATA \
    --remove_unused_columns false \
    --label_names "labels" \
    --prediction_loss_only false \
    --bf16 false \
    --bf16_full_eval false \
    --fp16 true \
    --fp16_full_eval true \
    --do_train \
    --do_eval \
    --tune_vision true \
    --tune_llm false \
    --use_lora true \
    --lora_target_modules "llm\..*layers\.\d+\.self_attn\.(q_proj|k_proj|v_proj|o_proj)" \
    --model_max_length $MODEL_MAX_Length \
    --max_slice_nums 9 \
    --max_steps 10000 \
    --eval_steps 1000 \
    --output_dir output/output__lora \
    --logging_dir output/output_lora \
    --logging_strategy "steps" \
    --per_device_train_batch_size 1 \
    --per_device_eval_batch_size 1 \
    --gradient_accumulation_steps 1 \
    --evaluation_strategy "steps" \
    --save_strategy "steps" \
    --save_steps 1000 \
    --save_total_limit 10 \
    --learning_rate 1e-6 \
    --weight_decay 0.1 \
    --adam_beta2 0.95 \
    --warmup_ratio 0.01 \
    --lr_scheduler_type "cosine" \
    --logging_steps 1 \
    --gradient_checkpointing true \
    --deepspeed ds_config_zero2.json \
    --report_to "tensorboard"
